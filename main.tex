% ── Cover pages (before \frontmatter, no page numbers) ──────
% Page anchors are switched off across this block. It runs on the default
% arabic numbering and each titlingpage resets the counter, so hyperref
% would emit page.1/page.2 anchors here; being first, they win, and
% \mainmatter's real page.1/page.2 get dropped as duplicates. Bibliography
% back-references to body pages 1 and 2 then land on a blank cover page.
% Nothing links to these pages, so dropping their anchors costs nothing.
\hypersetup{pageanchor=false}
\subfile{frontmatter/half_title}
\subfile{frontmatter/dedication}

\subfile{frontmatter/title_page}
\clearpage

% ── Front matter ─────────────────────────────────────────────
\frontmatter
% \frontmatter has flushed the cover pages, so anchors can come back on;
% the roman folios from here are unique against \mainmatter's arabic ones.
\hypersetup{pageanchor=true}

\subfile{frontmatter/acknowledgments}

% ── Appendices ───────────────────────────────────────────────
\appendix
\createmark{chapter}{left}{shownumber}{\appendixname\ }{.\space}
% Divider line in the ToC: an unnumbered part-level heading, with no page
% number, above the appendix entries (which memoir otherwise sets exactly
% like chapters). Written with \addtocontents rather than \addcontentsline
% so hyperref adds no part-level PDF bookmark, which would nest the
% Bibliography and the back matter under it.
\cleardoublepage
\hypertarget{appendices}{}
\addtocontents{toc}{\protect\contentsline{part}{\appendixtocname}{}{appendices}}
\subfile{appendix/appendix_A}
\subfile{appendix/appendix_B}

% Bibliography policy:
% - publications.bib: printed in full (P1, P2, ...) at the beginning.
% - publications_not.bib: printed in full (N1, N2, ...) right after, so any
%   in-text \cite of one of these (e.g. from a chapter mentioning related
%   work) resolves properly; also listed in the preface outline.
% - bibliography.bib: printed at the end, cited entries only, citation order.
% - code packages (selected by citekey in preamble/bibliography.tex): C1, C2, ...
